-- Insecure username-only coffee tracker
-- This stores coffees by username text, with open read/write policies.

create table if not exists public.coffees_by_username (
  id uuid primary key default gen_random_uuid(),
  username text not null,
  timestamp timestamptz not null default now()
);

-- Coffee Radar: temporary social coffee status
create table if not exists public.coffee_radar (
  username text primary key,
  coffee_at timestamptz not null,
  expires_at timestamptz not null,
  updated_at timestamptz not null default now()
);

alter table public.coffee_radar enable row level security;
drop policy if exists "read coffee radar" on public.coffee_radar;
drop policy if exists "insert coffee radar" on public.coffee_radar;
drop policy if exists "update coffee radar" on public.coffee_radar;
drop policy if exists "delete coffee radar" on public.coffee_radar;
create policy "read coffee radar" on public.coffee_radar for select using (true);
create policy "insert coffee radar" on public.coffee_radar for insert with check (true);
create policy "update coffee radar" on public.coffee_radar for update using (true) with check (true);
create policy "delete coffee radar" on public.coffee_radar for delete using (true);


-- Shared coffee events
create table if not exists public.shared_coffee_events (
  id uuid primary key default gen_random_uuid(),
  created_by text not null,
  created_at timestamptz not null default now()
);
create table if not exists public.shared_coffee_participants (
  event_id uuid not null references public.shared_coffee_events(id) on delete cascade,
  username text not null,
  primary key (event_id, username)
);
alter table public.shared_coffee_events enable row level security;
alter table public.shared_coffee_participants enable row level security;
drop policy if exists "read shared coffee events" on public.shared_coffee_events;
drop policy if exists "insert shared coffee events" on public.shared_coffee_events;
create policy "read shared coffee events" on public.shared_coffee_events for select using (true);
create policy "insert shared coffee events" on public.shared_coffee_events for insert with check (true);
drop policy if exists "read shared coffee participants" on public.shared_coffee_participants;
drop policy if exists "insert shared coffee participants" on public.shared_coffee_participants;
create policy "read shared coffee participants" on public.shared_coffee_participants for select using (true);
create policy "insert shared coffee participants" on public.shared_coffee_participants for insert with check (true);

create or replace function public.log_shared_coffee(p_created_by text, p_usernames text[])
returns uuid language plpgsql security definer set search_path=public as $$
declare v_event_id uuid; v_names text[]; v_count int;
begin
  select array_agg(name order by name), count(*) into v_names, v_count
  from (select distinct btrim(x) name from unnest(p_usernames) x where btrim(x) <> '') s;
  if v_count < 2 then raise exception 'At least two unique participants required'; end if;
  if not (p_created_by = any(v_names)) then raise exception 'Creator must be participant'; end if;
  insert into public.shared_coffee_events(created_by) values(p_created_by) returning id into v_event_id;
  insert into public.shared_coffee_participants(event_id,username) select v_event_id, unnest(v_names);
  insert into public.coffees_by_username(username) select unnest(v_names);
  return v_event_id;
end; $$;
grant execute on function public.log_shared_coffee(text,text[]) to anon, authenticated;

create index if not exists coffees_by_username_user_ts_idx
  on public.coffees_by_username (username, timestamp desc);

alter table public.coffees_by_username enable row level security;

-- Re-runnable: drop policies if they already exist
drop policy if exists "read all coffees" on public.coffees_by_username;
drop policy if exists "insert coffees" on public.coffees_by_username;
drop policy if exists "update coffees" on public.coffees_by_username;
drop policy if exists "delete coffees" on public.coffees_by_username;

-- Open policies (INSECURE): anyone can read/insert/delete/update.
create policy "read all coffees"
  on public.coffees_by_username for select
  using (true);

create policy "insert coffees"
  on public.coffees_by_username for insert
  with check (true);

create policy "update coffees"
  on public.coffees_by_username for update
  using (true)
  with check (true);

create policy "delete coffees"
  on public.coffees_by_username for delete
  using (true);

-- Optional: leaderboard function (aggregates by username)
create or replace function public.leaderboard_username_since(ts timestamptz)
returns table (username text, coffees int)
language sql
security definer
as $$
  select c.username, count(*)::int as coffees
  from public.coffees_by_username c
  where c.timestamp >= ts
  group by c.username
  order by coffees desc;
$$;

grant execute on function public.leaderboard_username_since(timestamptz) to anon, authenticated;
